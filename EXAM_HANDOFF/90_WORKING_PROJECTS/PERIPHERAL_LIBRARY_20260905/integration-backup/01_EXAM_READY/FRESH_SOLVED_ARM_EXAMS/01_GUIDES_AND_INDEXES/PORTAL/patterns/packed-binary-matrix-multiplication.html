How to recognize it

Matrix product over a defined binary operation.

When not to use it

Ordinary integer matrix multiply may use wrong algebra.

Inputs and outputs

State algebra, dimensions, and bit layout before coding.

Solving method

TEXT
1. Decode row/column bits
2. combine pairs
3. reduce with required XOR or OR
4. repack result

Worked trace

GF(2): [1,1] dot[1,1]=0, unlike Boolean-OR result1.

C approach

Make the contract explicit in types and parameters. State algebra, dimensions, and bit layout before coding.

Assembly considerations

Map this contract to registers, memory widths, and the frame before translating the steps. Ordinary integer matrix multiply may use wrong algebra.

Adaptation checklist

State algebra, dimensions, and bit layout before coding.

Boundary tests

identity; zero; distinction XOR versus OR.

Common mistakes

Ordinary integer matrix multiply may use wrong algebra. Use the trace above to locate the first differing step.

                PUSH    {R4-R11}
                MOV     R4, R0
                MOV     R5, R1
                MOV     R6, R2
                MOV     R7, R3
                MOVS    R8, #0
                MOVS    R9, #0
                MOVS    R10, #0         ; row
matmul_row
                MOVS    R11, #0         ; column
matmul_column
                MOVS    R0, #0          ; parity accumulator
                MOVS    R12, #0         ; k
matmul_dot
                LSL     R1, R10, #3
                ADD     R1, R1, R12     ; a index = row*8+k
                CMP     R1, #32
                BLO     matmul_a_low
                SUB     R2, R1, #32
                LSR     R2, R5, R2
                B       matmul_a_ready
matmul_a_low
                LSR     R2, R4, R1
matmul_a_ready
                AND     R2, R2, #1

                LSL     R1, R12, #3
                ADD     R1, R1, R11     ; b index = k*8+column
                CMP     R1, #32
                BLO     matmul_b_low
                SUB     R3, R1, #32
                LSR     R3, R7, R3
                B       matmul_b_ready
matmul_b_low
                LSR     R3, R6, R1
matmul_b_ready
                AND     R3, R3, #1
                AND     R2, R2, R3
                EOR     R0, R0, R2
                ADD     R12, R12, #1
                CMP     R12, #8
                BNE     matmul_dot

                CMP     R0, #0
                BEQ     matmul_next_column
                LSL     R1, R10, #3
                ADD     R1, R1, R11
                MOVS    R2, #1
                CMP     R1, #32
                BLO     matmul_set_low
                SUB     R1, R1, #32
                LSL     R2, R2, R1
                ORR     R9, R9, R2
                B       matmul_next_column
matmul_set_low
                LSL     R2, R2, R1
                ORR     R8, R8, R2
matmul_next_column
                ADD     R11, R11, #1
                CMP     R11, #8
                BNE     matmul_column
                ADD     R10, R10, #1
                CMP     R10, #8
                BNE     matmul_row
                MOV     R0, R8
                MOV     R1, R9
                POP     {R4-R11}
                BX      LR
                ENDP

                LTORG
                ALIGN   2
                END

#include <limits.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

/*
 * Exam-study reference: Packed binary matrix multiplication.
 * Recognition cue: multiply packed binary matrices over GF(2).
 *
 * Contract rules:
 * - Fixed-width types make width and signedness part of the interface.
 * - A pointer never carries its length; count/capacity arguments are explicit.
 * - const input objects are not mutated. Non-const outputs may be changed only
 *   within their documented bounds.
 * - Invalid, empty, duplicate and arithmetic-limit behavior is executable in
 *   pattern_edge_vectors() and described in the adjacent README.
 *
 * Trace the validation step first, then the main loop/recurrence invariant,
 * then the final result or capacity check. Public suffix functions are named
 * variants of the same advertised pattern, not unrelated shortcuts.
 */

/* Primary algorithm and its named variants. */
uint64_t algorithm_algorithm_pat_alg_packed_matmul_001_c(uint64_t a, uint64_t b) {
  uint64_t o = 0;
  uint32_t r, c, k;
  for (r = 0; r < 8; r++)
    for (c = 0; c < 8; c++) {
      uint32_t v = 0;
      for (k = 0; k < 8; k++)
        v ^= (uint32_t)((a >> (r * 8 + k)) & (b >> (k * 8 + c)) & 1u);
      o |= (uint64_t)v << (r * 8 + c);
    }
  return o;
}
