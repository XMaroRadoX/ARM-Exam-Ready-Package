How to recognize it

Summarize signed values in one pass.

When not to use it

Initializing max to zero fails all-negative inputs.

Inputs and outputs

Return values versus output pointers must match prototype.

Solving method

TEXT
1. Handle empty contract
2. initialize min/max from first element
3. widen sum if required
4. scan remainder

Worked trace

[-5,-2,-9] -> min-9,max-2,sum-16.

C approach

Make the contract explicit in types and parameters. Return values versus output pointers must match prototype.

Assembly considerations

Map this contract to registers, memory widths, and the frame before translating the steps. Initializing max to zero fails all-negative inputs.

Adaptation checklist

Return values versus output pointers must match prototype.

Boundary tests

empty; one; allnegative; sum overflow.

Common mistakes

Initializing max to zero fails all-negative inputs. Use the trace above to locate the first differing step.

/*
 * Exam-study reference: Signed min max and sum reduction.
 * Recognition cue: reduce array to minimum maximum and sum.
 *
 * Contract rules:
 * - Fixed-width types make width and signedness part of the interface.
 * - A pointer never carries its length; count/capacity arguments are explicit.
 * - const input objects are not mutated. Non-const outputs may be changed only
 *   within their documented bounds.
 * - Invalid, empty, duplicate and arithmetic-limit behavior is executable in
 *   pattern_edge_vectors() and described in the adjacent README.
 *
 * Trace the validation step first, then the main loop/recurrence invariant,
 * then the final result or capacity check. Public suffix functions are named
 * variants of the same advertised pattern, not unrelated shortcuts.
 */

/* Primary algorithm and its named variants. */

typedef struct {
  int32_t minimum;
  int32_t maximum;
  int64_t sum;
} algorithm_algorithm_pat_alg_reduction_001_assembly_signed_result_t;

typedef struct {
  uint32_t minimum;
  uint32_t maximum;
  uint64_t sum;
} algorithm_algorithm_pat_alg_reduction_001_assembly_unsigned_result_t;

int algorithm_algorithm_pat_alg_reduction_001_assembly(const int32_t *values, uint32_t count,
                          algorithm_algorithm_pat_alg_reduction_001_assembly_signed_result_t *result);

int algorithm_algorithm_pat_alg_reduction_001_assembly_unsigned(
    const uint32_t *values, uint32_t count,
    algorithm_algorithm_pat_alg_reduction_001_assembly_unsigned_result_t *result);



static int pattern_core_vector(void) {
  int32_t a[] = {-2, 5, 1};
  algorithm_algorithm_pat_alg_reduction_001_assembly_signed_result_t r;
  return algorithm_algorithm_pat_alg_reduction_001_assembly(a, 3, &r) && r.minimum == -2 && r.maximum == 5 &&
         r.sum == 4;
}
static int pattern_edge_vectors(void) {
  int32_t a[] = {INT32_MIN, INT32_MAX};
  uint32_t u[] = {0, UINT32_MAX};
  algorithm_algorithm_pat_alg_reduction_001_assembly_signed_result_t s;
  algorithm_algorithm_pat_alg_reduction_001_assembly_unsigned_result_t r;
  return !algorithm_algorithm_pat_alg_reduction_001_assembly(NULL, 0, &s) &&
         algorithm_algorithm_pat_alg_reduction_001_assembly(a, 2, &s) && s.sum == -1 &&
         algorithm_algorithm_pat_alg_reduction_001_assembly_unsigned(u, 2, &r) && r.sum == UINT32_MAX;
}
int pattern_test_suite(void) {
  if (!pattern_core_vector())
    return __LINE__;
  if (!pattern_edge_vectors())
    return __LINE__;
  return 0;
}
int test_main(void) { return pattern_test_suite(); }

volatile int pattern_result;
int main(void){exam_init();pattern_result=test_main();exam_led_write(pattern_result?0xFFu:0x01u);for(;;){}}

typedef struct {

int algorithm_algorithm_pat_alg_reduction_001_c(const int32_t *values, uint32_t count,
                          algorithm_algorithm_pat_alg_reduction_001_c_signed_result_t *result) {
  uint32_t index;

  if ((values == NULL) || (result == NULL) || (count == 0u)) {
    return 0;
  }
  result->minimum = values[0];
  result->maximum = values[0];
  result->sum = 0;
  for (index = 0u; index < count; ++index) {
    if (values[index] < result->minimum) {
      result->minimum = values[index];
    }
    if (values[index] > result->maximum) {
      result->maximum = values[index];
    }
    result->sum += values[index];
  }
  return 1;
}
