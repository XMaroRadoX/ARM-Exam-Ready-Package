How to recognize it

Exchange row and column positions in 64 packed bits.

When not to use it

Byte reversal alone is not transpose.

Inputs and outputs

Bit order and row packing must match the source contract.

Solving method

TEXT
1. Fix bit layout
2. extract bit(r,c)
3. place at(c,r)
4. preserve input if required

Worked trace

only bit(1,3)=1 -> only bit(3,1)=1.

C approach

Make the contract explicit in types and parameters. Bit order and row packing must match the source contract.

Assembly considerations

Map this contract to registers, memory widths, and the frame before translating the steps. Byte reversal alone is not transpose.

Adaptation checklist

Bit order and row packing must match the source contract.

Boundary tests

allzero; allone; identity; single bit at every position.

Common mistakes

Byte reversal alone is not transpose. Use the trace above to locate the first differing step.

; Handwritten Cortex-M3 Thumb exam reference.
; Verification status is supplied by the simulator/build reports.
; uint64_t algorithm_algorithm_pat_mem_packed_transpose_001_assembly(uint64_t packed)
; R0:R1=input, R0:R1=output. Leaf. R4-R9 saved (24-byte frame).
                AREA    |.text.patterns|, CODE, READONLY
                PRESERVE8
                THUMB
                EXPORT  algorithm_algorithm_pat_mem_packed_transpose_001_assembly

algorithm_algorithm_pat_mem_packed_transpose_001_assembly PROC
                PUSH    {R4-R9}
                MOV     R4, R0
                MOV     R5, R1
                MOVS    R6, #0
                MOVS    R7, #0
                MOVS    R8, #0          ; source bit index 0..63
transpose_loop
                CMP     R8, #32
                BLO     transpose_source_low
                SUB     R1, R8, #32
                LSR     R2, R5, R1
                B       transpose_have_bit
transpose_source_low
                LSR     R2, R4, R8
transpose_have_bit
                TST     R2, #1
                BEQ     transpose_next
                AND     R0, R8, #7      ; source column
                LSR     R1, R8, #3      ; source row
                LSL     R0, R0, #3
                ADD     R9, R0, R1      ; destination = column*8 + row
                MOVS    R2, #1
                CMP     R9, #32
                BLO     transpose_set_low
                SUB     R9, R9, #32
                LSL     R2, R2, R9
                ORR     R7, R7, R2
                B       transpose_next
transpose_set_low
                LSL     R2, R2, R9
                ORR     R6, R6, R2
transpose_next
                ADDS    R8, R8, #1
                CMP     R8, #64
                BNE     transpose_loop
                MOV     R0, R6
                MOV     R1, R7
                POP     {R4-R9}
                BX      LR
                ENDP

                LTORG
                ALIGN   2
                END

#include <limits.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

/*
 * Exam-study reference: Packed 8x8 bit-matrix transpose.
 * Recognition cue: transpose packed binary matrix rows and columns.
 *
 * Contract rules:
 * - Fixed-width types make width and signedness part of the interface.
 * - A pointer never carries its length; count/capacity arguments are explicit.
 * - const input objects are not mutated. Non-const outputs may be changed only
 *   within their documented bounds.
 * - Invalid, empty, duplicate and arithmetic-limit behavior is executable in
 *   pattern_edge_vectors() and described in the adjacent README.
 *
 * Trace the validation step first, then the main loop/recurrence invariant,
 * then the final result or capacity check. Public suffix functions are named
 * variants of the same advertised pattern, not unrelated shortcuts.
 */

/* Primary algorithm and its named variants. */
uint64_t algorithm_algorithm_pat_mem_packed_transpose_001_c(uint64_t x) {
  uint64_t y = 0;
  uint32_t r, c;
  for (r = 0; r < 8; r++)
    for (c = 0; c < 8; c++)
      y |= ((x >> (r * 8u + c)) & 1u) << (c * 8u + r);
  return y;
}
