Question 1 (10 points) 
Run-Length Encoding (RLE) is a simple, lossless data compression algorithm that reduces file size 
by replacing consecutive identical digits (a "run") with a single instance of the digit and a count of 
its repetitions. For example, the number 2222779 is encoded as 247291 because: 
 2 is repeated 4 times → 24 
 7 is repeated 2 times → 72 
 9 is repeated 1 time → 91 
 
Write the run_length_encoding subroutine in ARM assembly language. You must follow this 
prototype: 
 
uint32_t run_length_encoding(uint32_t digits); 
 
 
The subroutine receives an unsigned integer in input, and returns the encoded sequence of the run-
length-encoding algorithm as an unsigned integer. 
You can assume that: 
- each digit in the input number occurs at most 9 times 
- no overflow occurs during the computation 
- the encoded sequence can be held in a 32 bit unsigned integer. 
 
Suggestions. You can extract the digits of a number from least significant to most significant by 
repeatedly dividing by 10 and taking the remainder. For example: 
2222779 / 10 = 222277 with remainder 9 
222277 / 10 = 22227 with remainder 7 
22227 / 10 = 2222 with remainder 7 
2222 / 10 = 222 with remainder 2 
222 / 10 = 22 with remainder 2 
22 / 10 = 2 with remainder 2 
2 / 10 = 0 with remainder 2 
You know that there are no more digits when the quotient (result of the division) is 0. 
 
Note that the digits are extracted from right to left; then, they must be processed from left to right in 
order to compute the Run-Length Encoding. For each group of consecutive identical digits, first 
append the digit and then the count by repeated multiplication by 10. In detail, you start by 
initializing n = 0 and in every new iteration nnew = nold * 10 + digit and then nnew = nold * 10 + count. 
In the example: 
- the digit 2 occurs four times: 
o n = 0 * 10 + 2 = 2 
o n = 2 * 10 + 4 = 24 
- the digit 7 occurs twice: 
o n = 24 * 10 + 7 = 247 
o n = 247 * 10 + 2 = 2472 
- the digit 9 occurs once: 
o n = 2472 * 10 + 9 = 24729 
o n = 24729 * 10 + 1 = 247291

                AREA    |.text|, CODE, READONLY
                PRESERVE8
                THUMB

                EXPORT  run_length_encoding

; uint32_t run_length_encoding(uint32_t digits)
run_length_encoding PROC
                PUSH    {R4-R10, LR}
                SUB     SP, SP, #16     ; Reversed decimal digits.
                MOV     R4, SP
                MOV     R5, R0
                MOVS    R6, #0          ; Number of digits.
                MOVS    R10, #10

extract_digits
                UDIV    R7, R5, R10
                MLS     R8, R7, R10, R5
                STRB    R8, [R4, R6]
                ADDS    R6, R6, #1
                MOV     R5, R7
                CMP     R5, #0
                BNE     extract_digits

                SUBS    R6, R6, #1
                LDRB    R7, [R4, R6]    ; Current digit.
                MOVS    R8, #1          ; Run length.
                MOVS    R9, #0          ; Result.

look_loop
                CMP     R6, #0
                BEQ     append_look_run
                SUBS    R6, R6, #1
                LDRB    R5, [R4, R6]
                CMP     R5, R7
                BNE     append_then_continue
                ADDS    R8, R8, #1
                B       look_loop

append_then_continue
                MUL     R9, R9, R10
                ADD     R9, R9, R7
                MUL     R9, R9, R10
                ADD     R9, R9, R8
                MOV     R7, R5
                MOVS    R8, #1
                B       look_loop

append_look_run
                MUL     R9, R9, R10
                ADD     R9, R9, R7
                MUL     R9, R9, R10
                ADD     R9, R9, R8
                MOV     R0, R9
                ADD     SP, SP, #16
                POP     {R4-R10, PC}
                ENDP

                LTORG
                ALIGN   4
                END

/* Exam of 3 February 2026: debounced button and stable result display. */
#include <stdint.h>
#include "exam_api.h"
#include "LPC17xx.h"

extern uint32_t run_length_encoding(uint32_t digits);

/* Foreground-owned state: IRQs only capture input or publish events. */
static uint8_t latest_value;
static uint8_t have_sample;

void EINT1_IRQHandler(void)
{
  (void)exam_debounce_begin(EXAM_BUTTON_KEY1);
}

void RIT_IRQHandler(void)
{
  exam_rit_ack();
  exam_debounce_tick();
}

void ADC_IRQHandler(void) { exam_adc_irq_capture(); }

int main(void)
{
  exam_init();
  exam_buttons_init();
  /* Package debounce recommendation; the paper does not prescribe 10/50 ms. */
  if (exam_debounce_config(10u, 50u) != EXAM_OK ||
      exam_rit_config_ms(10u) != EXAM_OK) {
    exam_led_write(0xFFu);
    for (;;) { __WFI(); }
  }
  exam_rit_start();
  exam_adc_init();
  exam_adc_start();

  for (;;) {
    uint16_t sample;
    if (exam_adc_take(&sample)) {
      uint8_t value = (uint8_t)(sample >> 4);
      /* Hold a computed result until the potentiometer high8 value changes.
         The paper specifies no fixed result-hold duration. */
      if (!have_sample || value != latest_value) exam_led_write(value);
      latest_value = value;
      have_sample = 1u;
      exam_adc_start();
    }
    if ((exam_button_events_take() & EXAM_BUTTON_EVENT_KEY1) != 0u && have_sample) {
      exam_led_write((uint8_t)run_length_encoding(latest_value));
    }
    __WFI();
  }
}
