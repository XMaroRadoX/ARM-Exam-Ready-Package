Question 1 (6 points) 
A linear congruential generator (LCG) is an algorithm that produces a sequence of pseudo-random 
numbers. Write the LCGsequence subroutine, in ARM assembly, that returns the n-th element of 
a sequence, computed as follows: 
Xn = ((aXn-1 + c) EXOR (Xn-1 >> s)) mod m 
where >> s indicates the operation of logical shift right by s positions. 
 
The subroutine receives in input (in the specified order): 
1. the previous element of the sequence Xn-1 
2. the multiplier a 
3. the increment c 
4. the shift s 
5. the module m 
 
In order to test the procedure, define an array of bytes with DIM elements in a readwrite area of 
memory (DIM is a constant that can be freely chosen). Then, write a loop of DIM iterations in the 
Reset_Handler to fill the array. In each iteration of the loop, you call the LCGsequence 
subroutine with the appropriate parameters.

Question 3 (6 points) 
Extend the project to implement a rhythm game using the joystick on the board. The game will 
involve moving the joystick in response to the leds being switched on. 
When a led lights up, the player must move the joystick accordingly: 
- led 4: move the joystick up 
- led 5: move the joystick left 
- led 6: move the joystick right 
- led 7: move the joystick down 
Only the first movement of the joystick has to be considered (further movements within the 2.5-
second interval are ignored). If the movement corresponds to the correct direction for the lit led, 
increment the variable hit, otherwise increment miss. Regardless of the correctness, after the 
joystick movement the led is immediately switched off. 
The game continues for 10 sequence elements. After 10 elements, the game ends and the player's 
score is evaluated. If hit > miss, the player wins: turn on led 10 to indicate victory. Otherwise, 
the player looses: turn on led 11 to indicate defeat.

/* Exam of 1 July 2025, ARM2, questions 1 to 3. */
#include <stdint.h>
#include "exam_api.h"
#include "LPC17xx.h"

/* Single-LED API arguments are physical board labels LD4 through LD11. */

extern uint32_t LCGsequence(uint32_t previous,
                            uint32_t multiplier,
                            uint32_t increment,
                            uint32_t shift,
                            uint32_t modulus);

static uint32_t previous_value = 6u;
static uint32_t iteration;
static uint32_t expected_direction;
static uint32_t movement_seen;
static uint32_t num_correct;
static uint32_t num_wrong;
static uint32_t previous_joystick;

void TIMER1_IRQHandler(void)
{
  uint32_t remainder;

  if ((exam_timer_ack(EXAM_TIMER1) & 1u) == 0u) {
    return;
  }

  exam_led_clear();
  if (iteration >= 10u) {
    exam_timer_stop(EXAM_TIMER1);
    if (num_correct > num_wrong) {
      (void)exam_led_on(10u);
    } else {
      (void)exam_led_on(11u);
    }
    return;
  }

  previous_value = LCGsequence(previous_value, 157u, 3u, 3u, 256u);
  remainder = previous_value & 3u;

  movement_seen = 0u;
  if (remainder == 0u) {
    (void)exam_led_on(4u);
    expected_direction = EXAM_JOY_UP;
  } else if (remainder == 1u) {
    (void)exam_led_on(5u);
    expected_direction = EXAM_JOY_LEFT;
  } else if (remainder == 2u) {
    (void)exam_led_on(6u);
    expected_direction = EXAM_JOY_RIGHT;
  } else {
    (void)exam_led_on(7u);
    expected_direction = EXAM_JOY_DOWN;
  }

  iteration++;
}

static void joystick_event(uint32_t pressed)
{
  if (movement_seen || (pressed == 0u) || (iteration == 0u)) {
    return;
  }

  movement_seen = 1u;
  if (pressed & expected_direction) {
    num_correct++;
  } else {
    num_wrong++;
  }
  exam_led_clear();
}

void RIT_IRQHandler(void)
{
  uint32_t current;
  uint32_t pressed;
  exam_rit_ack();
  current=exam_joystick_read();
  pressed=exam_joystick_pressed_edges(previous_joystick,current);
  previous_joystick=current;
  joystick_event(pressed);
}

int main(void)
{
  exam_init();
  exam_joystick_init();
  previous_joystick=exam_joystick_read();
  if(exam_rit_config_ms(10u)==EXAM_OK)exam_rit_start();
  if(exam_timer_config_ms(EXAM_TIMER1,2500u,EXAM_TIMER_PERIODIC)==EXAM_OK)
    exam_timer_start(EXAM_TIMER1);

  for (;;) {
    __WFI();
  }
}
