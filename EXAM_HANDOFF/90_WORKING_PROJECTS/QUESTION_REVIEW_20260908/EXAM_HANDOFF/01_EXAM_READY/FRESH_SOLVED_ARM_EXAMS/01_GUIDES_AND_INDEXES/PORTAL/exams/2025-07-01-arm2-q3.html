Question · 2025-07-01 · ARM 2 LCG Rhythm

Algorithms: Games

Hardware: LED · Timer1 · RIT · Joystick

Interrupts: Timer IRQ · Acknowledgement

Timing: Delays

Review complete within the stated contract and limitations

Exact contract

LED4/5/6/7 expects UP/LEFT/RIGHT/DOWN. Count the first response per2500 ms window in hit/miss, clear the LED immediately, and keep the tenth window open for its full duration. Final hit>miss lights LED10; otherwise LED11.

Original paper — page 2

Solution reasoning

  1. Reuse Q2’s generator and open one response window per displayed value.
  2. Poll directional edges, ignoring centre and repeated movement after an answer.
  3. Close the last window, stop Timer1 and RIT, and show the final result without another generator call.

Worked trace

  1. First value 177 displays LED5; LEFT produces hit1.
  2. A wrong first move increments miss, and a later correct move in that window is ignored.
  3. After ten unanswered windows, hit0=miss0 and LED11 signals defeat.

/* Q3: complete game, using the Q1 assembly function.
 * Copy to Source/sample.c in a WORKING template.
 * Timer0 and RIT handlers are included here. Use both accompanying IRQ
 * replacement files so the linker sees exactly one owner per handler.
 */
#include "LPC17xx.h"
#include "exam_api.h"

extern uint32_t LCGsequence(uint32_t previous, uint32_t a,
                               uint32_t c, uint32_t n, uint32_t m);

/* Persistent sequence state. n is the NEXT index, from 0 through 9. */
static uint32_t previous = 6u;
static uint32_t n = 0u;

/* 1 accepts one answer; 0 ignores answers until the next round. */
static volatile uint32_t waiting_for_move = 0u;
static volatile uint32_t expected_direction = 0u;
volatile uint32_t hit = 0u;
volatile uint32_t miss = 0u;

/* Used to distinguish a new press from a direction being held. */
static uint32_t old_joystick = 0u;

/* remainder 0/1/2/3 -> LED 4/5/6/7 -> UP/LEFT/RIGHT/DOWN. */
static const uint32_t directions[4] = {
    EXAM_JOY_UP, EXAM_JOY_LEFT, EXAM_JOY_RIGHT, EXAM_JOY_DOWN
};

/* Called every 2.5 seconds by Timer0. */
void game_next_round(void)
{
    uint32_t value;
    uint32_t remainder;

    waiting_for_move = 0u;                /* Close the old response window. */
    exam_led_clear();                     /* Also clear an unanswered LED. */

    /* The tenth LED has already had its FULL 2.5-second window.
     * This is timer interrupt 11, but it does NOT call the LCG.
     */
    if (n == 10u) {
        exam_timer_stop(EXAM_TIMER1);
        exam_rit_stop();
        if (hit > miss) {
            exam_led_on(10u);              /* Victory. */
        } else {
            exam_led_on(11u);              /* Defeat, including a tie. */
        }
        return;
    }

    value = LCGsequence(previous, 157u, 3u, 3u, 256u);
    previous = value;                     /* Input for the next generation. */
    remainder = value % 4u;
    expected_direction = directions[remainder];
    exam_led_on((uint8_t)(4u + remainder));
    waiting_for_move = 1u;                /* Allow precisely one response. */
    n++;
}

/* Called every 10 ms by RIT; this is input sampling, not a new round. */
void game_poll_joystick(void)
{
    uint32_t current;
    uint32_t pressed;

    /* Ignore centre/select; retain only the four directional inputs. */
    current = exam_joystick_read()
            & (EXAM_JOY_UP | EXAM_JOY_LEFT | EXAM_JOY_RIGHT | EXAM_JOY_DOWN);
    pressed = exam_joystick_pressed_edges(old_joystick, current);

    /* Update even when answers are disabled. A held direction must not
     * become a new answer just because the next round has started.
     */
    old_joystick = current;
    if (waiting_for_move == 0u || pressed == 0u) {
        return;
    }

    waiting_for_move = 0u;                /* First movement consumes answer. */
    if (pressed == expected_direction) {
        hit++;
    } else {
        miss++;
    }
    exam_led_clear();                     /* Correct OR wrong: LED goes off. */
}

void TIMER1_IRQHandler(void)
{
    uint32_t pending = exam_timer_ack(EXAM_TIMER1);
    if ((pending & 1u) != 0u) {            /* MR0 is our 2.5-second interval. */
        game_next_round();
    }
}

void RIT_IRQHandler(void)
{
    exam_rit_ack();                       /* Acknowledge before sampling. */
    game_poll_joystick();
}

int main(void)
{
    exam_init();
    exam_led_clear();
    exam_joystick_init();
    old_joystick = exam_joystick_read()
                 & (EXAM_JOY_UP | EXAM_JOY_LEFT
                    | EXAM_JOY_RIGHT | EXAM_JOY_DOWN);

    if (exam_timer_config_ms(EXAM_TIMER1, 2500u,
                             EXAM_TIMER_PERIODIC) != EXAM_OK) {
        while (1) { }
    }
    if (exam_rit_config_ms(10u) != EXAM_OK) {
        while (1) { }
    }

    /* Equal priorities serialize the two handlers' shared-state updates.
     * volatile alone would NOT prevent one handler preempting the other.
     */
    NVIC_SetPriority(TIMER1_IRQn, 0u);
    NVIC_SetPriority(RIT_IRQn, 0u);
    exam_rit_start();
    exam_timer_start(EXAM_TIMER1);
    while (1) {
        __WFI();
    }
}

; uint32_t LCGsequence(previous, a, c, shift, modulus)
LCGsequence     PROC
                LDR     R12, [SP]       ; Fifth argument: modulus.
                LSR     R3, R0, R3      ; previous >> shift.
                MUL     R0, R0, R1
                ADDS    R0, R0, R2
                EORS    R0, R0, R3
                UDIV    R1, R0, R12
                MLS     R0, R1, R12, R0
                BX      LR
                ENDP

                LTORG
                ALIGN   4
                END

Code explanation

  • The saved waiting flag prevents post-game RIT activity from erasing the result.
  • Equal Timer1/RIT priorities serialize score and round transitions. The counters use the paper’s hit/miss names.

Complexity and memory

Constant work per poll and round.

Relevant mistakes

  • The previous implementation could still accept movement after displaying the final result.
  • A correct bit inside a diagonal input should not silently count as a uniquely correct direction.

Paper interpretation and limits

  • No movement is not counted as a miss because the paper only increments counters on movement. Centre/select is ignored; multiple new directional edges count as wrong. Physical input latency remains unverified.

Review findings and corrections

  • Corrected first-response ownership, ignored select, retained the full tenth window, stopped RIT at completion, and used equal timer/RIT priorities for shared state.
