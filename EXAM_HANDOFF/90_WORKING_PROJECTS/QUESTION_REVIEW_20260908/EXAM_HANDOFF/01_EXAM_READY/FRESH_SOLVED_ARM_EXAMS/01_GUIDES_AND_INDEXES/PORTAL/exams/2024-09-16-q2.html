Two successive raw button interrupts select increment and offset, then call Kruskal immediately on the second choice.

Question · 2024-09-16 · Kruskal Buttons

Algorithms: Recurrence · Graph

Hardware: LED · Buttons

Interrupts: EINT · Acknowledgement

ABI and memory: C-to-assembly

Known correctness limitation

  • Some valid button pairs expose the printed algorithm’s convergence defect; see Q1. The C answer lights all LEDs if any nonzero component labels remain after the call. This error indication is an implementation extension.
  • Starting another two-button attempt reinitializes the maze. Raw-button bounce and physical response still require hardware checks.

Review complete — unresolved paper-algorithm limitation

Exact contract

Initialize a 3x4 maze and wall arrays. The first button chooses increment: INT0=2, KEY1=3, KEY2=4. The second chooses offset with the same mapping and immediately calls kruskal with seven arguments.

Original paper — page 4

Solution reasoning

  1. Use selection_count to distinguish first from second input.
  2. Initialize each cell label from its row-major index and use wall marker2 at the appropriate borders.
  3. Pass rows before columns, then increment before offset.

Worked trace

  1. KEY2 then INT0 selects y4, x2 and produces the paper’s17-iteration example.
  2. After the first press alone, kruskal has not run.
  3. The finished maze contains only component label 0; border walls retain marker2.

Complete C answer

C
#include <stdint.h>
#include "exam_api.h"
#include "LPC17xx.h"

#define ROWS 3u
#define COLS 4u
extern void kruskal(uint8_t *maze,uint8_t *horizontal,uint8_t *vertical,
                    uint32_t rows,uint32_t columns,uint32_t increment,uint32_t offset);
static uint8_t maze[ROWS*COLS];
static uint8_t horizontal[ROWS*COLS];
static uint8_t vertical[ROWS*COLS];
static uint32_t increment;
static uint8_t selection_count;

static uint32_t button_value(exam_button_t b)
{ return b==EXAM_BUTTON_INT0?2u:(b==EXAM_BUTTON_KEY1?3u:4u); }

static void initialize_arrays(void)
{
  uint32_t r,c,k;
  for(r=0u;r<ROWS;++r)for(c=0u;c<COLS;++c){
    k=r*COLS+c; maze[k]=(uint8_t)k;
    horizontal[k]=(c==COLS-1u)?2u:1u;
    vertical[k]=(r==ROWS-1u)?2u:1u;
  }
}

static void handle_button(exam_button_t button)
{
  uint32_t value;
  value=button_value(button);
  if(selection_count==0u){initialize_arrays();increment=value;selection_count=1u;}
  else {
    uint32_t i;
    selection_count=0u;
    kruskal(maze,horizontal,vertical,ROWS,COLS,increment,value);
    /* The printed algorithm can stall for some button pairs. A nonzero
       component label means it returned a partial maze, not success. */
    for(i=0u;i<ROWS*COLS;++i) {
      if(maze[i]!=0u){exam_led_write(0xFFu);break;}
    }
  }
}

void EINT0_IRQHandler(void)
{ exam_button_ack(EXAM_BUTTON_INT0); handle_button(EXAM_BUTTON_INT0); }
void EINT1_IRQHandler(void)
{ exam_button_ack(EXAM_BUTTON_KEY1); handle_button(EXAM_BUTTON_KEY1); }
void EINT2_IRQHandler(void)
{ exam_button_ack(EXAM_BUTTON_KEY2); handle_button(EXAM_BUTTON_KEY2); }

int main(void)
{
  exam_init();
  initialize_arrays(); selection_count=0u; exam_led_clear();
  exam_buttons_init();

  for (;;) {
    __WFI();
  }
}

; void kruskal(uint8_t *maze,uint8_t *horizontal,uint8_t *vertical,
;              uint32_t rows,uint32_t columns,uint32_t y,uint32_t x)
kruskal         PROC
                MOV     R12, SP
                PUSH    {R4-R11, LR}
                SUB     SP, SP, #4      ; aligned 40-byte frame
                LDR     R4, [R12]       ; columns (argument 5)
                LDR     R5, [R12, #4]   ; increment y
                LDR     R6, [R12, #8]   ; offset x
                MOV     R7, R0          ; maze
                MOV     R8, R1          ; horizontal walls
                MOV     R9, R2          ; vertical walls
                MUL     R10, R3, R4     ; total cells
kr_loop         ADD     R6, R6, R5
                CMP     R6, R10
                BHS     kr_vertical
                LDRB    R0, [R8, R6]
                CMP     R0, #0
                BNE     kr_h_wall
                ADDS    R5, R5, #1
                B       kr_check_done
kr_h_wall       CMP     R0, #1
                BNE     kr_check_done
                ADD     R11, R6, #1
                LDRB    R2, [R7, R6]
                LDRB    R3, [R7, R11]
                CMP     R2, R3
                BEQ     kr_check_done
                MOVS    R0, #0
                STRB    R0, [R8, R6]
                B       kr_merge
kr_vertical     SUB     R6, R6, R10
                CMP     R6, R10
                BLO     kr_vertical_valid
                CMP     R5, R10         ; no later wall is reachable if x>=N,y>=N
                BHS     kr_done         ; failure: leave partial labels for caller
                B       kr_check_done
kr_vertical_valid
                LDRB    R0, [R9, R6]
                CMP     R0, #0
                BNE     kr_v_wall
                ADDS    R5, R5, #1
                B       kr_check_done
kr_v_wall       CMP     R0, #1
                BNE     kr_check_done
                ADD     R11, R6, R4
                LDRB    R2, [R7, R6]
                LDRB    R3, [R7, R11]
                CMP     R2, R3
                BEQ     kr_check_done
                MOVS    R0, #0
                STRB    R0, [R9, R6]
kr_merge        CMP     R2, R3
                ITE     LO
                MOVLO   R0, R2          ; m
                MOVHS   R0, R3
                ITE     LO
                MOVLO   R1, R3          ; n
                MOVHS   R1, R2
                MOVS    R11, #0
kr_replace      CMP     R11, R10
                BHS     kr_check_done
                LDRB    R2, [R7, R11]
                CMP     R2, R1
                IT      EQ
                STRBEQ  R0, [R7, R11]
                ADDS    R11, R11, #1
                B       kr_replace
kr_check_done   MOVS    R11, #0
kr_nonzero      CMP     R11, R10
                BHS     kr_done
                LDRB    R0, [R7, R11]
                CMP     R0, #0
                BNE     kr_loop
                ADDS    R11, R11, #1
                B       kr_nonzero
kr_done         ADD     SP, SP, #4
                POP     {R4-R11, PC}
                ENDP
                END

Code explanation

  • C supplies the last three arguments on the stack automatically.
  • The first parameter is the maze pointer, not the row count used by other maze routines.

Complexity and memory

O(1) selection plus the Q1 wall-removal work.

Relevant mistakes

  • Swapping the two chosen button values changes the generated maze.
  • The INT0 handler is required as well as KEY1 and KEY2.

Paper interpretation and limits

  • Some valid button pairs expose the printed algorithm’s convergence defect; see Q1. The C answer lights all LEDs if any nonzero component labels remain after the call. This error indication is an implementation extension.
  • Starting another two-button attempt reinitializes the maze. Raw-button bounce and physical response still require hardware checks.

Review findings and corrections

  • Reinitializes every new two-button attempt and displays all LEDs on when the returned labels show an incomplete maze.
