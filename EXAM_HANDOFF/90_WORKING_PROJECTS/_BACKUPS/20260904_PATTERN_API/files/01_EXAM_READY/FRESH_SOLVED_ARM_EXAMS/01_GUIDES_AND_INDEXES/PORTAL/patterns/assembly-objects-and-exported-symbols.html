How to recognize it

C and ARM share named data/function storage.

When not to use it

EXPORT is not a C definition and extern allocates no storage.

Inputs and outputs

Validate structure offsets instead of guessing padding.

Solving method

TEXT
1. Choose one writable/read-only definition
2. export/import names
3. match widths/alignment
4. reference address correctly

Worked trace

DCD7 exported object -> extern uint32_t sees7.

C approach

Make the contract explicit in types and parameters. Validate structure offsets instead of guessing padding.

Assembly considerations

Map this contract to registers, memory widths, and the frame before translating the steps. EXPORT is not a C definition and extern allocates no storage.

Adaptation checklist

Validate structure offsets instead of guessing padding.

Boundary tests

name mismatch; duplicate storage; const write.

Common mistakes

#include <limits.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

/*
 * Exam-study reference: Assembly objects and exported symbols.
 * Recognition cue: define constants initialized reserved aligned exported data.
 *
 * Contract rules:
 * - Fixed-width types make width and signedness part of the interface.
 * - A pointer never carries its length; count/capacity arguments are explicit.
 * - const input objects are not mutated. Non-const outputs may be changed only
 *   within their documented bounds.
 * - Invalid, empty, duplicate and arithmetic-limit behavior is executable in
 *   pattern_edge_vectors() and described in the adjacent README.
 *
 * Trace the validation step first, then the main loop/recurrence invariant,
 * then the final result or capacity check. Public suffix functions are named
 * variants of the same advertised pattern, not unrelated shortcuts.
 */

/* Primary algorithm and its named variants. */

enum { pat_data_asm_objects_001_BUFFER_WORDS = 8 };

const uint32_t pat_data_asm_objects_001_magic = UINT32_C(0x13579BDF);
uint32_t pat_data_asm_objects_001_initialized_words[4] = {1u, 2u, 3u, 4u};
uint32_t
    pat_data_asm_objects_001_workspace[pat_data_asm_objects_001_BUFFER_WORDS];
const uint32_t pat_data_asm_objects_001_masks[4] = {1u, 2u, 4u, 8u};

uint32_t pat_data_asm_objects_001(const uint32_t *values, uint32_t count) {
  uint32_t sum = 0u;

  if ((values == NULL) && (count != 0u)) {
    return 0u;
  }

  while (count != 0u) {
    sum += *values;
    ++values;
    --count;
  }

  return sum;
}

#ifdef PATTERN_HOST_TEST
static int pattern_core_vector(void) {
  uint32_t a[] = {1, 2, 3};
  return pat_data_asm_objects_001(a, 3) == 6;
}
static int pattern_edge_vectors(void) {
  if(pat_data_asm_objects_001_magic!=0x13579BDFu||pat_data_asm_objects_001_initialized_words[3]!=4||pat_data_asm_objects_001_masks[3]!=8)return 0;
  uint32_t one[] = {UINT32_MAX};
  return pat_data_asm_objects_001(NULL, 0) == 0u &&
         pat_data_asm_objects_001(one, 1) == UINT32_MAX;
}
int pattern_test_suite(void) {
  if (!pattern_core_vector())
    return __LINE__;
  if (!pattern_edge_vectors())
    return __LINE__;
  return 0;
}
int main(void) { return pattern_test_suite(); }
#endif
